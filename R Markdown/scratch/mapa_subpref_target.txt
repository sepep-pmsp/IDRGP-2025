gerar_mapa_subpref <- function(nome_sp) {
  # 4.1) Filtros internos
  subpref_ativa <- todas_subprefs |> dplyr::filter(subprefeitura == nome_sp)
  distritos_internos <- distritos_sf |> dplyr::filter(subprefeitura_geo == nome_sp)
  row_perfil <- df_perfil |> dplyr::filter(subprefeitura == nome_sp)

  if (nrow(row_perfil) != 1) {
    warning("Perfil nao encontrado (ou duplicado) para a subprefeitura: ", nome_sp, ". Usando perfil padrão.", call. = FALSE)
    row_perfil <- tibble::tibble(
      subprefeitura = nome_sp,
      sigla = dplyr::coalesce(subpref_ativa$sigla[1], ""),
      status = "Dentro do IDRGP Alvo",
      idrgp_real = NA_real_
    )
  }

  cor_sp <- unname(cores_status[row_perfil$status])
  if (is.na(cor_sp)) {
    cor_sp <- "#FF00FF" # fallback chamativo para diagnosticar status nao mapeado
    warning("Status sem cor na paleta (cores_status) para: ", nome_sp, " | status = ", row_perfil$status)
  }

  sigla_sp <- row_perfil$sigla

  # 4.2) Coluna de rótulo dos distritos (ordem mandatória)
  col_label <- NULL
  if ("sg_distrito_municipal" %in% names(distritos_internos)) {
    col_label <- "sg_distrito_municipal"
  } else {
    candidatos <- c("sigla", "nm_distrito", "nm_distr", "distrito", "nome")
    achada <- intersect(names(distritos_internos), candidatos)[1]
    if (!is.na(achada) && length(achada) > 0) col_label <- achada
  }

  if (is.null(col_label)) {
    warning("Nenhuma coluna de rótulo de distrito encontrada para ", nome_sp, ". Mapa sairá sem rótulos de distrito.")
  }

  # 4.3) Camadas do ggplot (ordem exata e mandatória) -----------------------------
  p <- ggplot2::ggplot() +
    # 1) Fundo do município (cinza denso, sem bordas)
    ggplot2::geom_sf(data = todas_subprefs, fill = "#B0B0B0", color = NA) +
    # 2) Borda externa do município (contorno SP)
    ggplot2::geom_sf(data = sf::st_union(todas_subprefs), fill = NA, color = "gray30", linewidth = 0.6) +
    # 3) Siglas de todas as subprefeituras
    ggplot2::geom_sf_text(
      data = todas_subprefs,
      ggplot2::aes(label = sigla),
      size = 2.8, color = "gray20", fontface = "bold"
    ) +
    # 4) Preenchimento da subprefeitura ativa (cor do status, sem borda)
    ggplot2::geom_sf(data = subpref_ativa, fill = cor_sp, color = NA, alpha = 0.9) +
    # 5) Limites dos distritos internos (tracejado sutil)
    ggplot2::geom_sf(data = distritos_internos, fill = NA, color = "gray75", linetype = "dashed", linewidth = 0.35) +
    # 6) Rótulos dos distritos internos (com check_overlap = TRUE)
    {
      if (!is.null(col_label)) {
        ggplot2::geom_sf_text(
          data = distritos_internos,
          ggplot2::aes(label = .data[[col_label]]),
          size = 2, color = "gray30", check_overlap = TRUE
        )
      }
    } +
    # Tema e textos
    ggplot2::labs(
      title = paste0("Subprefeitura ", nome_sp, " (", sigla_sp, ")"),
      subtitle = paste0("IDRGP: ", row_perfil$idrgp_real, " | Status: ", row_perfil$status)
    ) +
    ggplot2::theme_void() +
    ggplot2::coord_sf(
      xlim = bbox_sp[c(1, 3)],
      ylim = bbox_sp[c(2, 4)],
      expand = TRUE
    )

  # Legenda discreta de cores (somente se houver múltiplos status no "universo" analisado)
  # Nota: neste mapa há apenas 1 subprefeitura destacada, entÃ£o a legenda só é útil quando
  # se deseja comunicar a paleta completa.
  if (dplyr::n_distinct(df_perfil$status) > 1) {
    legend_sf <- df_perfil |>
      dplyr::distinct(status) |>
      dplyr::mutate(x = bbox_sp$xmin, y = bbox_sp$ymin) |>
      sf::st_as_sf(coords = c("x", "y"), crs = 4326)

    p <- p +
      ggplot2::geom_sf(
        data = legend_sf,
        ggplot2::aes(fill = status),
        shape = 21,
        alpha = 0
      ) +
      ggplot2::scale_fill_manual(values = cores_status, name = "Status IDRGP", limits = names(cores_status))
  }

  # Norte e escala (minimalistas; escala 10 km)
  if (!requireNamespace("ggspatial", quietly = TRUE)) {
    warning("Pacote `ggspatial` não instalado; mapa será gerado sem norte/escala.", call. = FALSE)
  } else {
    p <- p +
      ggspatial::annotation_north_arrow(
        location = "tr",
        style = ggspatial::north_arrow_minimal
      ) +
      ggspatial::annotation_scale(
        location = "br",
        width_hint = 0.3,
        style = "ticks",
        unit_category = "metric",
        text_cex = 0.7
      )
  }

  return(p)
}
