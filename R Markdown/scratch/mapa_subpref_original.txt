gerar_mapa_subpref <- function(nome_sp) {
  # 6.1) Filtros internos
  subpref_ativa <- todas_subprefs |> dplyr::filter(subprefeitura == nome_sp)
  distritos_internos <- distritos_sf |> dplyr::filter(subprefeitura_geo == nome_sp)
  row_perfil <- df_perfil |> dplyr::filter(subprefeitura == nome_sp)

  if (nrow(row_perfil) != 1) {
    stop("Perfil não encontrado (ou duplicado) para a subprefeitura: ", nome_sp)
  }

  cor_sp <- unname(cores_status[row_perfil$status])
  if (is.na(cor_sp)) {
    cor_sp <- "#FF00FF"
    warning("Status sem cor na paleta (cores_status) para: ", nome_sp, " | status = ", row_perfil$status)
  }

  sigla_sp <- row_perfil$sigla

  # 6.2) Coluna de rótulo dos distritos
  col_label <- NULL
  if ("sg_distrito_municipal" %in% names(distritos_internos)) {
    col_label <- "sg_distrito_municipal"
  } else {
    candidatos <- c("sigla", "nm_distrito", "nm_distr", "distrito", "nome")
    achada <- intersect(names(distritos_internos), candidatos)[1]
    if (!is.na(achada) && length(achada) > 0) col_label <- achada
  }

  if (is.null(col_label)) {
    warning("Nenhuma coluna de rótulo de distrito encontrada para ", nome_sp, ". Mapa sairá sem rótulos de distrito.")
  }

  # 6.3) Camadas do ggplot (na ordem EXATA) -------------------------------------
  p <- ggplot2::ggplot() +
    # 1) Fundo do município — cinza claro e suave (≈10%, não denso)
    ggplot2::geom_sf(data = todas_subprefs, fill = "#EAEAEA", color = "gray60", linewidth = 0.3) +
    # 2) Borda externa do município
    ggplot2::geom_sf(data = sf::st_union(todas_subprefs), fill = NA, color = "gray30", linewidth = 0.6) +
    # 3) Siglas de TODAS as subprefeituras — legíveis, negrito
    ggplot2::geom_sf_text(
      data = todas_subprefs,
      ggplot2::aes(label = sigla),
      size = 2.5, color = "gray10", fontface = "bold"
    ) +
    # 4) Subprefeitura ativa (status) com borda nítida e contínua
    ggplot2::geom_sf(data = subpref_ativa, fill = cor_sp, color = "gray20",
                     linewidth = 0.7, alpha = 0.9) +
    # 5) Limites dos distritos internos (tracejado leve)
    ggplot2::geom_sf(
      data = distritos_internos,
      fill = NA, color = "gray75", linetype = "dashed", linewidth = 0.35
    ) +
    # 6) Rótulos dos distritos internos
    {
      if (!is.null(col_label)) {
        ggplot2::geom_sf_text(
          data = distritos_internos,
          ggplot2::aes(label = .data[[col_label]]),
          size = 2, color = "gray30", check_overlap = TRUE
        )
      }
    } +
    # 7) Sigla da subprefeitura ativa em destaque (maior, sobre as demais)
    ggplot2::geom_sf_text(
      data = subpref_ativa,
      ggplot2::aes(label = sigla),
      size = 4.0, color = "white", fontface = "bold"
    ) +
    # 8) Título: apenas nome + sigla, centralizado; sem subtítulo
    ggplot2::labs(
      title = paste0("Subprefeitura ", nome_sp, " (", sigla_sp, ")")
    ) +
    # 9) Coordenadas geográficas nos eixos (NÃO remover coord_sf)
    ggplot2::coord_sf(
      xlim = bbox_sp[c(1, 3)],
      ylim = bbox_sp[c(2, 4)],
      expand = TRUE
    ) +
    # 10) Escala no canto inferior direito
    ggspatial::annotation_scale(
      location   = "br",
      width_hint = 0.3,
      style      = "ticks",
      text_cex   = 0.7
    ) +
    # 11) Seta norte logo acima da escala (pad_y empurra para cima)
    ggspatial::annotation_north_arrow(
      location    = "br",
      which_north = "true",
      height      = ggplot2::unit(1.2, "cm"),
      width       = ggplot2::unit(0.9, "cm"),
      pad_y       = ggplot2::unit(0.8, "cm"),
      style       = ggspatial::north_arrow_minimal()
    ) +
    # 12) Tema: minimal preserva coordenadas; remove apenas elementos supérfluos
    ggplot2::theme_minimal(base_size = 11) +
    ggplot2::theme(
      text             = ggplot2::element_text(family = "sans"),
      plot.title       = ggplot2::element_text(hjust = 0.5, size = 14, face = "bold"),
      panel.border     = ggplot2::element_rect(color = "black", fill = NA, linewidth = 0.8),
      panel.grid.major = ggplot2::element_line(color = "gray85", linewidth = 0.3),
      panel.grid.minor = ggplot2::element_blank(),
      axis.text.x      = ggplot2::element_text(size = 8),
      axis.text.y      = ggplot2::element_text(size = 8),
      axis.title       = ggplot2::element_blank(),
      legend.position  = "bottom",
      legend.title     = ggplot2::element_text(size = 9, face = "bold"),
      legend.text      = ggplot2::element_text(size = 8)
    )

  # Legenda discreta de status (se houver múltiplos status entre as 32 subprefs)
  if (dplyr::n_distinct(df_perfil$status) > 1) {
    p <- p +
      ggplot2::scale_fill_manual(values = cores_status, name = "Status IDRGP", limits = names(cores_status))
  }

  p
}
